def completar_historico(base_actual, base_anterior):

    indice_anterior = {}

    # Crear índice de la base anterior por Crédito
    for fila in base_anterior:
        credito = fila["Crédito"].strip()
        indice_anterior[credito] = fila

    # Pasar histórico a la base actual
    for fila in base_actual:

        credito = fila["Crédito"].strip()

        if credito in indice_anterior:

            anterior = indice_anterior[credito]

            fila["OB ANTERIOR"] = anterior["OB 3"]
            fila["OB DIC"] = anterior["OB MES"]
            fila["FECHA INICIO"] = anterior["FECHA INICIO"]
            fila["FECHA FN"] = anterior["FECHA FIN"]
            fila["CERTIFICADO"] = anterior["CERTIFICADO"]
            fila["POLIZA ZU"] = anterior["POLIZA ZU"]
            fila["OBS EDW"] = anterior["OBS EDW"]

    return base_actual
