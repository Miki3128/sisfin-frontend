import csv


R58 = "Bases/R58.csv"
BASE_ACTIVO_PF = "Bases/Base_Activo_PF.csv"


# ==============================
# LEER R58
# ==============================

nuevos_registros = []

with open(R58, "r", encoding="utf-8-sig", newline="") as archivo:
    lector = csv.DictReader(archivo)

    for fila in lector:

        # Filtrar únicamente ModalidadPF = Mensual
        if fila["ModalidadPF"].strip() != "Mensual":
            continue

        nuevo = {
            "CRÉDITO": fila["crédito"].strip(),
            "CEDULA": fila["Número identificación"].strip(),
            "FECHA DESEMBOLSO": fila["Fecha Desembolso"].strip(),
            "TIPO PF": "Mensual",
            "FIN VIGENCIA": "Pago Mensual",
            "ESTADO": "Si",
            "MORA": "0",
            "Estado Credito": "Vigente A"
        }

        nuevos_registros.append(nuevo)


# ==============================
# AGREGAR A LA BASE ACTIVO PF
# ==============================

encabezados = [
    "CRÉDITO",
    "CEDULA",
    "FECHA DESEMBOLSO",
    "TIPO PF",
    "FIN VIGENCIA",
    "ESTADO",
    "MORA",
    "Estado Credito"
]


with open(
    BASE_ACTIVO_PF,
    "a",
    encoding="utf-8-sig",
    newline=""
) as archivo:

    escritor = csv.DictWriter(
        archivo,
        fieldnames=encabezados
    )

    for registro in nuevos_registros:
        escritor.writerow(registro)


print(f"Registros agregados: {len(nuevos_registros)}")
