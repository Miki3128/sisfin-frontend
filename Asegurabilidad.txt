for fila in base_asegurabilidad:

    poliza_zurich = str(
        fila.get("POLIZA ZURICH", "")
    ).strip()

    if poliza_zurich in indice_productos:

        registro = indice_productos[poliza_zurich]

        modalidad = str(
            registro.get("PRIMA", "")
        ).strip()

        if modalidad:

            fila["Modalidad_Seguro_Vida"] = modalidad
            fila["MODALIDAD POLIZA ACTUAL 2"] = modalidad
            fila["MODALIDAD POLIZA ACTUAL"] = modalidad

            encontrados += 1

    else:

        modalidad_existente = str(
            fila.get("Modalidad_Seguro_Vida", "")
        ).strip()

        if modalidad_existente:

            fila["MODALIDAD POLIZA ACTUAL 2"] = modalidad_existente
            fila["MODALIDAD POLIZA ACTUAL"] = modalidad_existente

        no_encontrados += 1
