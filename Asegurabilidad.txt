def ajustar_saldo_menor(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Candado: si ya tiene OBS EDWARD, no tocar
        obs_edward = str(
            fila.get("OBS EDWARD", "")
        ).strip()

        if obs_edward:
            no_procesados += 1
            continue

        # Leer SALDO
        saldo_texto = str(
            fila.get("SALDO", "")
        ).strip()

        if not saldo_texto:
            no_procesados += 1
            continue

        try:
            saldo = float(
                saldo_texto.replace(".", "").replace(",", ".")
            )
        except ValueError:
            no_procesados += 1
            continue

        # Solo procesar saldos menores a $1.000.000
        if saldo < 1000000:

            fila["OB 3"] = "Saldo menor"
            fila["OB MES"] = "Politica - Saldo menor"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE SALDO MENOR ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
