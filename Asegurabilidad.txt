import openpyxl


def guardar_excel_con_cancelados(ruta, base_asegurabilidad, cancelados):

    libro = openpyxl.Workbook()

    # Hoja principal
    hoja_principal = libro.active
    hoja_principal.title = "Base Asegurabilidad"

    if base_asegurabilidad:

        columnas = list(base_asegurabilidad[0].keys())

        for columna_numero, columna in enumerate(columnas, start=1):
            hoja_principal.cell(
                row=1,
                column=columna_numero,
                value=columna
            )

        for fila_numero, fila in enumerate(base_asegurabilidad, start=2):

            for columna_numero, columna in enumerate(columnas, start=1):

                hoja_principal.cell(
                    row=fila_numero,
                    column=columna_numero,
                    value=fila.get(columna, "")
                )

    # Hoja Cancelado
    hoja_cancelado = libro.create_sheet("Cancelado")

    if cancelados:

        columnas_cancelado = list(cancelados[0].keys())

        for columna_numero, columna in enumerate(
            columnas_cancelado,
            start=1
        ):

            hoja_cancelado.cell(
                row=1,
                column=columna_numero,
                value=columna
            )

        for fila_numero, fila in enumerate(cancelados, start=2):

            for columna_numero, columna in enumerate(
                columnas_cancelado,
                start=1
            ):

                hoja_cancelado.cell(
                    row=fila_numero,
                    column=columna_numero,
                    value=fila.get(columna, "")
                )

    libro.save(ruta)

    print(
        f"\nBase Asegurabilidad Nueva guardada correctamente: {ruta}"
    )
