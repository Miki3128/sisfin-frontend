def ajustar_endosada(base_asegurabilidad, año, mes):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Primero: solo los que tienen Endoso en OB DIC
        ob_dic = str(
            fila.get("OB DIC", "")
        ).strip().lower()

        if ob_dic != "endoso":
            no_procesados += 1
            continue

        fecha_fin_texto = str(
            fila.get("FECHA FN", "")
        ).strip()

        if not fecha_fin_texto:
            no_procesados += 1
            continue

        try:
            fecha_fin = datetime.strptime(
                fecha_fin_texto,
                "%Y-%m-%d"
            )

        except ValueError:

            try:
                fecha_fin = datetime.strptime(
                    fecha_fin_texto,
                    "%d/%m/%Y"
                )

            except ValueError:
                no_procesados += 1
                continue

        # La fecha debe ser desde el mes que estamos procesando
        # en adelante
        if (
            fecha_fin.year > año
            or (
                fecha_fin.year == año
                and fecha_fin.month >= mes
            )
        ):

            fila["OB 3"] = "Endosada"
            fila["OB MES"] = fila.get("OB DIC", "")
            fila["FECHA INICIO"] = fila.get("FECH INICIO 2", "")
            fila["FECHA FIN"] = fila.get("FECHA FN", "")
            fila["CERTIFICADO"] = fila.get("CERTIFICADO 2", "")
            fila["POLIZA ZURICH"] = "-"
            fila["OBS EDWARD"] = fila.get("OBS EDWARD 2", "")

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE ENDOSADA ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
