def obtener_cancelados(base_asegurabilidad):

    cancelados = []

    for fila in base_asegurabilidad:

        estado = str(
            fila.get("Estado crèdito", "")
        ).strip().upper()

        if estado == "NO":

            fila_cancelada = fila.copy()

            # No modificar ninguna columna existente.
            # Solo agregar la nueva columna al final.
            fila_cancelada["cancelacion"] = "Cancelado"

            cancelados.append(fila_cancelada)

    print("\n=== CONTROL CANCELADOS ===")
    print(f"Registros cancelados: {len(cancelados)}")

    return cancelados
