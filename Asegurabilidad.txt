def cruzar_aumentos_capital(base_asegurabilidad, aumentos_capital):

    indice_aumentos = {}

    for fila in aumentos_capital:
        credito = str(
            fila.get("Crédito", "")
        ).strip()

        if credito:
            indice_aumentos[credito] = fila

    encontrados = 0
    no_encontrados = 0

    for fila in base_asegurabilidad:

        # Si OBS EDWARD ya tiene información, no tocar el registro
        if str(fila.get("OBS EDWARD", "")).strip():
            continue

        credito = str(
            fila.get("Crédito", "")
        ).strip()

        if credito in indice_aumentos:

            registro = indice_aumentos[credito]

            # Traer OB MES VIGENTE de AUMENTOS CAPITAL
            fila["OB MES VIGENTE"] = registro.get(
                "OB MES VIGENTE", ""
            ).strip()

            encontrados += 1

        else:
            no_encontrados += 1

    print("\n=== CRUCE AUMENTOS CAPITAL ===")
    print(f"Créditos encontrados: {encontrados}")
    print(f"Créditos no encontrados: {no_encontrados}")

    return base_asegurabilidad
