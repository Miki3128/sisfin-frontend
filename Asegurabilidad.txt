from datetime import datetime
import calendar


def validar_r21_vs_asegurabilidad(r21, base_asegurabilidad, año, mes):

    # Crear índice de créditos de la Base Asegurabilidad
    creditos_asegurabilidad = set()

    for fila in base_asegurabilidad:
        credito = str(fila["Crédito"]).strip()

        if credito:
            creditos_asegurabilidad.add(credito)

    # Último día del mes que estamos trabajando
    ultimo_dia = calendar.monthrange(año, mes)[1]

    fecha_limite = datetime(año, mes, ultimo_dia)

    encontrados = 0
    no_encontrados = []

    for fila in r21:

        vigente = str(fila["Vigente"]).strip().upper()

        # Solo créditos vigentes
        if vigente != "SI":
            continue

        credito = str(fila["NumCreditoBanco"]).strip()

        if not credito:
            continue

        # Validar fecha de desembolso
        fecha_texto = str(fila["Fecha Desembolso"]).strip()

        try:
            fecha_desembolso = datetime.strptime(
                fecha_texto,
                "%d/%m/%Y"
            )
        except ValueError:
            continue

        # No validar créditos posteriores al mes trabajado
        if fecha_desembolso > fecha_limite:
            continue

        # Buscar crédito en Base Asegurabilidad
        if credito in creditos_asegurabilidad:
            encontrados += 1
        else:
            no_encontrados.append({
                "Crédito": credito,
                "Fecha Desembolso": fecha_texto,
                "Vigente": vigente
            })

    print("\n=== CONTROL R21 VS BASE ASEGURABILIDAD ===")

    print(f"Créditos vigentes validados: {encontrados + len(no_encontrados)}")
    print(f"Encontrados en Base Asegurabilidad: {encontrados}")
    print(f"No encontrados: {len(no_encontrados)}")

    if no_encontrados:

        print("\n=== CRÉDITOS NO ENCONTRADOS ===")

        for fila in no_encontrados:
            print(
                fila["Crédito"],
                "-> Fecha:",
                fila["Fecha Desembolso"]
            )

    else:

        print("\nTodos los créditos vigentes están en la Base Asegurabilidad.")

    return no_encontrados

def separar_cancelados(base_asegurabilidad):

    base_activa = []
    cancelados = []

    for fila in base_asegurabilidad:

        estado = str(fila.get("Estado crèdito", "")).strip().upper()

        if estado == "NO":

            fila_cancelada = fila.copy()
            fila_cancelada["cancelacion"] = "Cancelado"

            cancelados.append(fila_cancelada)

        else:

            base_activa.append(fila)

    print("\n=== CONTROL CANCELADOS ===")
    print(f"Registros activos: {len(base_activa)}")
    print(f"Registros cancelados: {len(cancelados)}")

    return base_activa, cancelados
