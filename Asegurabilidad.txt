import csv

R58 = "Bases/R58.csv"
BASE_VIGENTES = "Vigentes/Base_Activo_PF_Vigentes.csv"

COLUMNAS = [
    "CRÉDITO",
    "CEDULA",
    "FECHA DESEMBOLSO",
    "TIPO PF",
    "FIN VIGENCIA",
    "ESTADO",
    "MORA",
    "Estado Credito"
]

registros = []

# ============================================================
# LEER R58
# ============================================================

with open(R58, "r", encoding="utf-8-sig", newline="") as archivo:
    lector = csv.DictReader(archivo, delimiter=";")

    for fila in lector:

        if fila["ModalidadPF"].strip() != "Mensual":
            continue

        registros.append([
            fila["crédito"].strip(),
            fila["Número identificación"].strip(),
            fila["Fecha Desembolso"].strip(),
            "Mensual",
            "Pago Mensual",
            "Si",
            "0",
            "Vigente A"
        ])


# ============================================================
# CREAR BASE VIGENTES
# ============================================================

with open(BASE_VIGENTES, "w", encoding="utf-8-sig", newline="") as archivo:

    # Esta línea le dice a Excel:
    # "el separador de este archivo es ;"
    archivo.write("sep=;\n")

    escritor = csv.writer(
        archivo,
        delimiter=";",
        lineterminator="\n"
    )

    # Encabezados
    escritor.writerow(COLUMNAS)

    # Registros
    escritor.writerows(registros)


print(f"Registros agregados: {len(registros)}")
print("Base Activo PF - Vigentes creada correctamente.")
