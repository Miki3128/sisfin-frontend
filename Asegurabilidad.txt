def ajustar_ventas_nuevas_vigentes(base_asegurabilidad, mes_venta_nueva):

    texto_buscar = f"Aseg {mes_venta_nueva}"

    for fila in base_asegurabilidad:

        obs_edward = str(
            fila.get("OBS EDWARD", "")
        ).strip()

        ob_anterior = str(
            fila.get("OB ANTERIOR", "")
        ).strip()

        # Solo registros que vienen de la base Vigentes
        if obs_edward == "Asegurado en base zurich":

            if texto_buscar.lower() in ob_anterior.lower():

                fila["OB 3"] = "Asegurado"

    return base_asegurabilidad
