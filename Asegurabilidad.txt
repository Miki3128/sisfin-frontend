def cruzar_vigentes(base_asegurabilidad, base_vigentes):

    indice_vigentes = {}

    for fila in base_vigentes:
        credito = str(fila.get("No Credito", "")).strip()

        if credito:
            indice_vigentes[credito] = fila

    encontrados = 0
    no_encontrados = 0

    for fila in base_asegurabilidad:

        credito = str(fila.get("Crédito", "")).strip()

        if credito in indice_vigentes:

            registro = indice_vigentes[credito]

            certificado = str(
                registro.get("NRO DOCUM", "")
            ).strip()

            if certificado:

                fila["OB 3"] = fila.get("OB ANTERIOR", "")
                fila["OB MES"] = fila.get("OB DIC", "")
                fila["CERTIFICADO"] = certificado
                fila["FECHA INICIO"] = registro.get("FECHA INI", "").strip()
                fila["FECHA FIN"] = registro.get("FECHA FIN", "").strip()
                fila["POLIZA ZURICH"] = registro.get("POL MAESTRA", "").strip()
                fila["OBS EDWARD"] = "Asegurado en base zurich"

                encontrados += 1

        else:
            no_encontrados += 1

    print("\n=== CRUCE VIGENTES ===")
    print(f"Créditos encontrados: {encontrados}")
    print(f"Créditos no encontrados: {no_encontrados}")

    return base_asegurabilidad
