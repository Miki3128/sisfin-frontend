def ajustar_poliza_vencida_endosos(base_asegurabilidad, año, mes):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Solo revisar los que quedaron como Endosada
        ob_mes = str(
            fila.get("OB MES", "")
        ).strip().lower()

        if ob_mes != "endosada":
            no_procesados += 1
            continue

        fecha_fin_texto = str(
            fila.get("FECHA FIN", "")
        ).strip()

        if not fecha_fin_texto:
            no_procesados += 1
            continue

        try:
            fecha_fin = datetime.strptime(
                fecha_fin_texto,
                "%Y-%m-%d"
            )

        except ValueError:

            try:
                fecha_fin = datetime.strptime(
                    fecha_fin_texto,
                    "%d/%m/%Y"
                )

            except ValueError:
                no_procesados += 1
                continue

        # Validar que FECHA FIN sea del mes que estamos procesando
        if (
            fecha_fin.year == año
            and fecha_fin.month == mes
        ):

            fila["CERTIFICADO"] = "Poliza Vencida"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE POLIZA VENCIDA ENDOSOS ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
