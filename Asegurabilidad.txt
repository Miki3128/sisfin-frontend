def ajustar_poliza_vigente_endosos(base_asegurabilidad, año, mes):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Solo revisar los que siguen como Endosada
        ob_mes = str(
            fila.get("OB MES", "")
        ).strip().lower()

        if ob_mes != "endosada":
            no_procesados += 1
            continue

        fecha_fin_texto = str(
            fila.get("FECHA FIN ENDOSOS", "")
        ).strip()

        if not fecha_fin_texto:
            no_procesados += 1
            continue

        try:
            fecha_fin_endosos = datetime.strptime(
                fecha_fin_texto,
                "%Y-%m-%d"
            )

        except ValueError:

            try:
                fecha_fin_endosos = datetime.strptime(
                    fecha_fin_texto,
                    "%d/%m/%Y"
                )

            except ValueError:
                no_procesados += 1
                continue

        # La fecha debe ser posterior al mes que estamos procesando
        if (
            fecha_fin_endosos.year > año
            or (
                fecha_fin_endosos.year == año
                and fecha_fin_endosos.month > mes
            )
        ):

            fila["CERTIFICADO"] = "Poliza Vigente"
            fila["FECHA INICIO"] = fila.get(
                "FECHA INICIO ENDOSOS", ""
            )
            fila["FECHA FIN"] = fila.get(
                "FECHA FIN ENDOSOS", ""
            )

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE POLIZA VIGENTE ENDOSOS ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
