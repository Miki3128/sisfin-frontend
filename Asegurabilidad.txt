def cruzar_vida_mensual_empleados(base_asegurabilidad, base_vida_mensual_empleados, periodo_mes):

    indice_vida_mensual_empleados = {}

    for fila in base_vida_mensual_empleados:
        credito = str(fila.get("Crédito", "")).strip()

        if credito:
            indice_vida_mensual_empleados[credito] = fila

    encontrados = 0
    no_encontrados = 0

    for fila in base_asegurabilidad:

        credito = str(fila.get("Crédito", "")).strip()

        if credito in indice_vida_mensual_empleados:

            registro = indice_vida_mensual_empleados[credito]

            certificado = str(
                registro.get("No Certificado VD", "")
            ).strip()

            if certificado:

                fila["MODALIDAD POLIZA ACTUAL"] = "Mensual"
                fila["OB 3"] = "Asegurado Mensual"
                fila["OB MES"] = "Asegurado"
                fila["CERTIFICADO"] = certificado
                fila["FECHA INICIO"] = registro.get("Fecha Envio", "").strip()
                fila["FECHA FIN"] = registro.get("Fecha Envio Fin", "").strip()
                fila["POLIZA ZURICH"] = registro.get("Poliza Matriz Vida", "").strip()
                fila["OBS EDWARD"] = f"Asegurado {periodo_mes} mensual"

                encontrados += 1

        else:
            no_encontrados += 1

    print("\n=== CRUCE VIDA MENSUAL EMPLEADOS ===")
    print(f"Créditos encontrados: {encontrados}")
    print(f"Créditos no encontrados: {no_encontrados}")

    return base_asegurabilidad
