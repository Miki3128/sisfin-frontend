def ajustar_renovar_aumentos(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        ob_mes_vigente = str(
            fila.get("OB MES VIGENTE", "")
        ).strip().lower()

        # Solo procesar los que cruzaron como Renovar
        if ob_mes_vigente != "renovar":
            no_procesados += 1
            continue

        fila["OB 3"] = "En renovacion anual"
        fila["OB MES"] = "Asegurado"
        fila["OBS EDWARD"] = "En renovacion Anual"

        procesados += 1

    print("\n=== AJUSTE RENOVAR AUMENTOS CAPITAL ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
