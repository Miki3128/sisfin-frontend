def cruzar_base_endosos(base_asegurabilidad, base_endosos):

    indice_endosos = {}

    for fila in base_endosos:

        credito = str(
            fila.get("No. Crédito", "")
        ).strip()

        if credito:
            indice_endosos[credito] = fila

    encontrados = 0
    no_encontrados = 0

    for fila in base_asegurabilidad:

        credito = str(
            fila.get("Crédito", "")
        ).strip()

        if credito in indice_endosos:

            registro = indice_endosos[credito]

            fila["FECHA INICIO ENDOSOS"] = registro.get(
                "FECHA INICIO", ""
            ).strip()

            fila["FECHA FIN ENDOSOS"] = registro.get(
                "FECHA FIN", ""
            ).strip()

            encontrados += 1

        else:
            no_encontrados += 1

    print("\n=== CRUCE BASE ENDOSOS ===")
    print(f"Créditos encontrados: {encontrados}")
    print(f"Créditos no encontrados: {no_encontrados}")

    return base_asegurabilidad
