def crear_nuevos(r58, r21):

    resultado = []

    # Crear índice de R21 por crédito
    indice_r21 = {}

    for fila in r21:
        credito = fila["NumCreditoBanco"].strip()
        indice_r21[credito] = fila

    for fila in r58:

        credito = fila["Crédito"].strip()

        # Solo créditos 830 y 834
        if not (credito.startswith("830") or credito.startswith("834")):
            continue

        # Buscar información del crédito en R21
        datos_r21 = indice_r21.get(credito)

        if datos_r21:
            estado_credito = datos_r21["Vigente"]
            mora = datos_r21["Dias Mora Historico"]
            saldo = datos_r21["Saldo"]
        else:
            estado_credito = ""
            mora = ""
            saldo = ""

        # Determinar tipo de préstamo y empresa
        if credito.startswith("830"):
            tipo_prestamo = "VEHICULO SANTANDER"
            empresa = "BCO"

        else:
            tipo_prestamo = "VEHICULO MAZ"
            empresa = "MAZ"

        nueva_fila = {

            "INICIAL": credito[:3],

            "TIPO PRESTAMO": tipo_prestamo,

            "EMPRESA": empresa,

            "Crédito": credito,

            "Cedula": fila["Número Identificación"],

            "Estado crèdito": estado_credito,

            "Mora": mora,

            "Solicitud": fila["Solicitud"],

            "Fecha Desembolso": fila["Fecha Desembolso"],

            "Nombres": fila["Nombres"],

            "Vir Seguro Vida Financiado": fila["Vlr Seguro Vida Financiado"],

            "Vir Seg Desempleo Single Premium Financiado": fila["Vlr Seg Desempleo Single Premium Financiado"],

            "SALDO": saldo,

            "ESTADO CREDITO": "-",

            "Fechafinal": fila["Fechafinal"],

            "Poliza Vida Endosada": fila["Poliza Vida Endosada"],

            "Modalidad_Seguro_Vida": fila["Modalidad_Seguro_Vida"],

            "Paga poliza Vida decontado": fila["Paga poliza Vida decontado"],

            "Tipo_Garantia": fila["Tipo_Garantia"],

            "RESTRUCTURADO": "NO"
        }

        resultado.append(nueva_fila)

    return resultado
