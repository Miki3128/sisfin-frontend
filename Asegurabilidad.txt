 def agregar_creditos_anteriores(base_nueva, base_anterior):

    # Créditos que ya están en la nueva base
    creditos_nuevos = set()

    for fila in base_nueva:
        credito = str(fila.get("Crédito", "")).strip()

        if credito:
            creditos_nuevos.add(credito)

    agregados = 0
    repetidos = 0

    for fila in base_anterior:

        credito = str(fila.get("Crédito", "")).strip()

        if not credito:
            continue

        # Si el crédito ya está en la nueva base, no lo agregamos
        if credito in creditos_nuevos:
            repetidos += 1
            continue

        # =====================================================
        # CREAR LA FILA CON EXACTAMENTE LAS COLUMNAS DE LA BASE
        # =====================================================

        nueva_fila = {
            "INICIAL": fila.get("INICIAL", ""),
            "TIPO PRESTAMO": fila.get("TIPO PRESTAMO", ""),
            "EMPRESA": fila.get("EMPRESA", ""),
            "Crédito": fila.get("Crédito", ""),
            "Cedula": fila.get("Cedula", ""),
            "Estado crèdito": fila.get("Estado crèdito", ""),
            "Mora": fila.get("Mora", ""),
            "Solicitud": fila.get("Solicitud", ""),
            "Fecha Desembolso": fila.get("Fecha Desembolso", ""),
            "Nombres": fila.get("Nombres", ""),
            "Vlr Seguro Vida Financiado": fila.get(
                "Vlr Seguro Vida Financiado", ""
            ),
            "Vlr Seg Desempleo Single Premium Financiado": fila.get(
                "Vlr Seg Desempleo Single Premium Financiado", ""
            ),
            "SALDO": fila.get("SALDO", ""),
            "ESTADO CREDITO": fila.get("ESTADO CREDITO", ""),
            "Fechafinal": fila.get("Fechafinal", ""),
            "Poliza Vida Endosada": fila.get("Poliza Vida Endosada", ""),
            "Modalidad_Seguro_Vida": fila.get(
                "Modalidad_Seguro_Vida", ""
            ),
            "Paga poliza Vida decontado": fila.get(
                "Paga poliza Vida decontado", ""
            ),
            "Tipo_Garantia": fila.get("Tipo_Garantia", ""),
            "RESTRUCTURADO": fila.get("RESTRUCTURADO", ""),

            # HISTÓRICO
            "MODALIDAD POLIZA ACTUAL 2": fila.get(
                "MODALIDAD POLIZA ACTUAL", ""
            ),
            "OB ANTERIOR": fila.get("OB 3", ""),
            "OB DIC": fila.get("OB MES", ""),
            "FECHA INICIO 2": fila.get("FECHA INICIO", ""),
            "FECHA FN": fila.get("FECHA FIN", ""),
            "CERTIFICADO 2": fila.get("CERTIFICADO", ""),
            "POLIZA ZURICH 2": fila.get("POLIZA ZURICH", ""),
            "OBS EDWARD 2": fila.get("OBS EDWARD", ""),

            # NUEVO PERÍODO: quedan vacías
            "MODALIDAD POLIZA ACTUAL": "",
            "OB 3": "",
            "OB MES": "",
            "FECHA INICIO": "",
            "FECHA FIN": "",
            "CERTIFICADO": "",
            "POLIZA ZURICH": "",
            "OBS EDWARD": ""
        }

        # Agregar el crédito completo
        base_nueva.append(nueva_fila)

        creditos_nuevos.add(credito)
        agregados += 1

    print("\n=== CRÉDITOS BASE ANTERIOR ===")
    print(f"Créditos ya existentes: {repetidos}")
    print(f"Créditos antiguos agregados: {agregados}")
    print(f"Total base nueva: {len(base_nueva)}")

    return base_nueva
