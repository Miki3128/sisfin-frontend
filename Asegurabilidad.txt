def ajustar_mora_mayor_90(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Candado: si ya tiene OBS EDWARD, no tocar
        obs_edward = str(
            fila.get("OBS EDWARD", "")
        ).strip()

        if obs_edward:
            no_procesados += 1
            continue

        # Solo procesar los que tengan Mora mayor a 90
        mora_texto = str(
            fila.get("Mora", "")
        ).strip()

        try:
            mora = float(
                mora_texto.replace(".", "").replace(",", ".")
            )
        except ValueError:
            no_procesados += 1
            continue

        if mora > 90:

            fila["OB 3"] = "Mora > 90"
            fila["OB MES"] = "Politica - Mora > 90"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE MORA MAYOR A 90 ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
