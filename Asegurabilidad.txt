import openpyxl


# ==============================
# RUTAS
# ==============================

R58 = "Bases/R58.xlsx"
BASE_VIGENTES = "Vigentes/Base_Activo_PF_Vigentes.xlsx"


# ==============================
# ABRIR ARCHIVOS
# ==============================

wb_r58 = openpyxl.load_workbook(R58, data_only=True)
ws_r58 = wb_r58.active

wb_vigentes = openpyxl.load_workbook(BASE_VIGENTES)
ws_vigentes = wb_vigentes.active


# ==============================
# IDENTIFICAR ENCABEZADOS R58
# ==============================

encabezados_r58 = {}

for columna in range(1, ws_r58.max_column + 1):
    encabezado = ws_r58.cell(row=1, column=columna).value

    if encabezado:
        encabezados_r58[str(encabezado).strip()] = columna


# ==============================
# COLUMNAS QUE NECESITAMOS
# ==============================

col_modalidad = encabezados_r58["ModalidadPF"]
col_credito = encabezados_r58["crédito"]
col_cedula = encabezados_r58["Número identificación"]
col_fecha = encabezados_r58["Fecha Desembolso"]


# ==============================
# RECORRER R58
# ==============================

registros_agregados = 0

for fila in range(2, ws_r58.max_row + 1):

    modalidad = ws_r58.cell(
        row=fila,
        column=col_modalidad
    ).value

    # Solo ModalidadPF = Mensual
    if str(modalidad).strip() != "Mensual":
        continue

    credito = ws_r58.cell(
        row=fila,
        column=col_credito
    ).value

    cedula = ws_r58.cell(
        row=fila,
        column=col_cedula
    ).value

    fecha_desembolso = ws_r58.cell(
        row=fila,
        column=col_fecha
    ).value

    # ==============================
    # AGREGAR A VIGENTES
    # ==============================

    ws_vigentes.append([
        credito,
        cedula,
        fecha_desembolso,
        "Mensual",
        "Pago Mensual",
        "Si",
        0,
        "Vigente A"
    ])

    registros_agregados += 1


# ==============================
# GUARDAR
# ==============================

wb_vigentes.save(BASE_VIGENTES)

print(f"Registros agregados: {registros_agregados}")
