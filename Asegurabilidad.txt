def ajustar_mora_mayor_90_aumentos(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        ob_mes_vigente = str(
            fila.get("OB MES VIGENTE", "")
        ).strip().lower()

        # Solo procesar los que cruzaron como Mora > 90
        if ob_mes_vigente != "mora > 90":
            no_procesados += 1
            continue

        mora_texto = str(
            fila.get("Mora", "")
        ).strip()

        try:
            mora = float(
                mora_texto.replace(".", "").replace(",", ".")
            )
        except ValueError:
            mora = 0

        # Validar que realmente tenga mora mayor a 90
        if mora > 90:

            fila["OB 3"] = "Mora > 90"
            fila["OB MES"] = "Politica - Mora > 90"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE MORA MAYOR A 90 AUMENTOS CAPITAL ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
