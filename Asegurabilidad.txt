def ajustar_castigados(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Candado: si ya tiene OBS EDWARD, no tocar
        obs_edward = str(
            fila.get("OBS EDWARD", "")
        ).strip()

        if obs_edward:
            no_procesados += 1
            continue

        # Buscar la palabra "castigado" en ESTADO CREDITO
        estado_credito = str(
            fila.get("ESTADO CREDITO", "")
        ).strip().lower()

        if "castigado" not in estado_credito:
            no_procesados += 1
            continue

        fila["OB 3"] = "Castigado"
        fila["OB MES"] = "Politica - castigado"
        fila["OBS EDWARD"] = "-"

        procesados += 1

    print("\n=== AJUSTE CASTIGADOS ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
