def ajustar_castigado_aumentos(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        ob_mes_vigente = str(
            fila.get("OB MES VIGENTE", "")
        ).strip().lower()

        if ob_mes_vigente == "castigado":

            fila["OB 3"] = "Castigado"
            fila["OB MES"] = "Politica - castigado"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE CASTIGADO AUMENTOS CAPITAL ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
