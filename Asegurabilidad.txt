def separar_cancelados(base_asegurabilidad):

    base_activa = []
    cancelados = []

    for fila in base_asegurabilidad:

        estado = str(
            fila.get("Estado crèdito", "")
        ).strip().upper()

        if estado == "NO":

            fila_cancelada = fila.copy()

            # Mantener Estado crèdito como NO
            fila_cancelada["Estado crèdito"] = "NO"

            # Agregar columna adicional al final
            fila_cancelada["cancelacion"] = "Cancelado"

            cancelados.append(fila_cancelada)

        else:

            base_activa.append(fila)

    print("\n=== CONTROL CANCELADOS ===")
    print(
        f"Registros que permanecen en la base: "
        f"{len(base_activa)}"
    )
    print(
        f"Registros enviados a Cancelado: "
        f"{len(cancelados)}"
    )

    return base_activa, cancelados
