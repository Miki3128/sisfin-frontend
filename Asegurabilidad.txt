def ajustar_endoso_no_cruzado(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # OB MES debe estar vacío
        ob_mes = str(
            fila.get("OB MES", "")
        ).strip()

        if ob_mes:
            no_procesados += 1
            continue

        # OB DIC debe ser Endosada
        ob_dic = str(
            fila.get("OB DIC", "")
        ).strip().lower()

        if ob_dic != "endosada":
            no_procesados += 1
            continue

        # Si llegó hasta aquí significa que no cruzó
        # con una póliza vigente de Base Endosos
        fila["OB 3"] = "Forzar"
        fila["OB MES"] = "Politica - forzar"
        fila["OBS EDWARD"] = "Forzar"

        procesados += 1

    print("\n=== AJUSTE ENDOSO NO CRUZADO ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
