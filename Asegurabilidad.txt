import csv

R58 = "Bases/R58.csv"
BASE_VIGENTES = "Vigentes/Base_Activo_PF_Vigentes.csv"

registros_agregados = []

# Leer R58
with open(R58, "r", encoding="utf-8-sig", newline="") as archivo:
    lector = csv.DictReader(archivo, delimiter=";")

    for fila in lector:

        if fila["ModalidadPF"].strip() != "Mensual":
            continue

        registros_agregados.append([
            fila["crédito"].strip(),
            fila["Número identificación"].strip(),
            fila["Fecha Desembolso"].strip(),
            "Mensual",
            "Pago Mensual",
            "Si",
            "0",
            "Vigente A"
        ])


# Agregar a Vigentes
with open(
    BASE_VIGENTES,
    "a",
    encoding="utf-8",
    newline=""
) as archivo:

    escritor = csv.writer(
        archivo,
        delimiter=";",
        lineterminator="\n"
    )

    for registro in registros_agregados:
        escritor.writerow(registro)


print(f"Registros agregados: {len(registros_agregados)}")
