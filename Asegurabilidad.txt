def ajustar_creditos_finalizados(base_asegurabilidad, año, mes):

    procesados = 0
    no_procesados = 0

    # Calcular mes anterior y mes siguiente
    mes_anterior = mes - 1
    año_anterior = año

    if mes_anterior == 0:
        mes_anterior = 12
        año_anterior -= 1

    mes_siguiente = mes + 1
    año_siguiente = año

    if mes_siguiente == 13:
        mes_siguiente = 1
        año_siguiente += 1

    meses_validos = [
        (año_anterior, mes_anterior),
        (año, mes),
        (año_siguiente, mes_siguiente)
    ]

    for fila in base_asegurabilidad:

        # Candado: si ya tiene OBS EDWARD, no tocar
        obs_edward = str(
            fila.get("OBS EDWARD", "")
        ).strip()

        if obs_edward:
            no_procesados += 1
            continue

        fecha_texto = str(
            fila.get("Fechafinal", "")
        ).strip()

        if not fecha_texto:
            no_procesados += 1
            continue

        try:
            fecha_final = datetime.strptime(
                fecha_texto,
                "%Y-%m-%d"
            )

        except ValueError:

            try:
                fecha_final = datetime.strptime(
                    fecha_texto,
                    "%d/%m/%Y"
                )

            except ValueError:
                no_procesados += 1
                continue

        periodo = (
            fecha_final.year,
            fecha_final.month
        )

        # Solo procesar junio, julio y agosto de 2026
        if periodo in meses_validos:

            fila["OB 3"] = "Credito finalizado"
            fila["OB MES"] = "Politica - credito finalizado"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE CREDITOS FINALIZADOS ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
