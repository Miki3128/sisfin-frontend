def ajustar_castigados(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Si OBS EDWARD ya tiene información, no tocar el registro
        if str(fila.get("OBS EDWARD", "")).strip():
            continue

        ob_dic = str(
            fila.get("OB DIC", "")
        ).strip().lower()

        estado_credito = str(
            fila.get("ESTADO CREDITO", "")
        ).strip().lower()

        # Solo procesar Politica - castigado
        # cuando el estado actual del crédito sigue siendo castigado
        if (
            ob_dic == "politica - castigado"
            and estado_credito == "castigado"
        ):

            fila["OB 3"] = "Castigado"
            fila["OB MES"] = "Politica - castigado"
            fila["POLIZA ZURICH"] = "Castigado"
            fila["OBS EDWARD"] = "-"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE CREDITOS CASTIGADOS ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
