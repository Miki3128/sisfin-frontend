def ajustar_mora_superada_aumentos(base_asegurabilidad):

    procesados = 0
    no_procesados = 0

    for fila in base_asegurabilidad:

        # Solo revisar los que salieron en el filtro Mora > 90
        ob_mes_vigente = str(
            fila.get("OB MES VIGENTE", "")
        ).strip().lower()

        if ob_mes_vigente != "mora > 90":
            no_procesados += 1
            continue

        mora_texto = str(
            fila.get("Mora", "")
        ).strip()

        try:
            mora = float(
                mora_texto.replace(".", "").replace(",", ".")
            )
        except ValueError:
            no_procesados += 1
            continue

        saldo = str(
            fila.get("SALDO", "")
        ).strip()

        # Si ya no está en mora y tiene saldo
        if mora < 90 and saldo:

            fila["OB 3"] = "Forzar"
            fila["OB MES"] = "Politica - forzar"
            fila["OBS EDWARD"] = "Forzar"

            procesados += 1

        else:
            no_procesados += 1

    print("\n=== AJUSTE MORA SUPERADA AUMENTOS CAPITAL ===")
    print(f"Créditos procesados: {procesados}")
    print(f"Créditos no procesados: {no_procesados}")

    return base_asegurabilidad
